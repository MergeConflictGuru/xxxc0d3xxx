cmake_minimum_required(VERSION 3.16)
project(nightcall_audio C CXX)
set(CMAKE_CXX_STANDARD 17)
set(OPUS_BUILD_SHARED_LIBRARY OFF CACHE BOOL "" FORCE)
set(OPUS_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(OPUS_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(OPUS_DRED OFF CACHE BOOL "" FORCE)
set(OPUS_OSCE OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_subdirectory(opus EXCLUDE_FROM_ALL)
if(ANDROID)
  target_compile_options(opus PRIVATE -O2)
endif()

option(NIGHTCALL_NETWORK "Build the compact data-only direct networking engine" ON)
if(NIGHTCALL_NETWORK)
  set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
  set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
  set(USE_SHARED_MBEDTLS_LIBRARY OFF CACHE BOOL "" FORCE)
  set(DISABLE_PACKAGE_CONFIG_AND_INSTALL OFF CACHE BOOL "" FORCE)
  set(MBEDTLS_USER_CONFIG_FILE "${CMAKE_CURRENT_SOURCE_DIR}/mbed_config.h" CACHE STRING "" FORCE)
  add_subdirectory(mbedtls EXCLUDE_FROM_ALL)
  add_library(MbedTLS::MbedTLS ALIAS mbedtls)
  set(USE_MBEDTLS ON CACHE BOOL "" FORCE)
  set(NO_WEBSOCKET ON CACHE BOOL "" FORCE)
  set(NO_MEDIA ON CACHE BOOL "" FORCE)
  set(NO_EXAMPLES ON CACHE BOOL "" FORCE)
  set(NO_TESTS ON CACHE BOOL "" FORCE)
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  add_subdirectory(datachannel EXCLUDE_FROM_ALL)
  add_library(nightcall_net SHARED net.cpp)
  target_link_libraries(nightcall_net PRIVATE datachannel-static)
  target_include_directories(nightcall_net PRIVATE datachannel/deps/json/include)
  if(ANDROID)
    foreach(component nightcall_net datachannel-static usrsctp juice-static mbedtls mbedx509 mbedcrypto)
      if(TARGET ${component})
        target_compile_options(${component} PRIVATE -O2)
      endif()
    endforeach()
  endif()
endif()
add_library(nightcall_audio SHARED audio.c)
target_link_libraries(nightcall_audio PRIVATE opus)
target_include_directories(nightcall_audio PRIVATE opus/include)
if(MSVC)
  target_compile_options(nightcall_audio PRIVATE /O2)
else()
  target_compile_options(nightcall_audio PRIVATE -O2 -fvisibility=hidden)
endif()
